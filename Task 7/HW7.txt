   1. DIM_TIME_DAY
   ============================================================ */

CREATE TABLE IF NOT EXISTS bl_dm.dim_time_day
(
    event_dt        DATE PRIMARY KEY,
    day_num         SMALLINT NOT NULL DEFAULT -1,
    month_num       SMALLINT NOT NULL DEFAULT -1,
    month_name      VARCHAR(20) NOT NULL DEFAULT 'n.a.',
    quarter_num     SMALLINT NOT NULL DEFAULT -1,
    year_num        SMALLINT NOT NULL DEFAULT -1
);


/* ============================================================
   2. DIM_CUSTOMERS_SCD
   ============================================================ */

CREATE TABLE IF NOT EXISTS bl_dm.dim_customers_scd
(
    customer_surr_id   BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    customer_src_id    VARCHAR(255) NOT NULL DEFAULT 'n.a.',
    gender             VARCHAR(50)  NOT NULL DEFAULT 'n.a.',
    age                INTEGER      NOT NULL DEFAULT -1,
    customer_type      VARCHAR(100) NOT NULL DEFAULT 'n.a.',
    start_dt           DATE         NOT NULL DEFAULT DATE '1900-01-01',
    end_dt             DATE         NOT NULL DEFAULT DATE '9999-12-31',
    is_active          CHAR(1)      NOT NULL DEFAULT 'Y',
    ta_insert_dt       TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ta_update_dt       TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    source_system      VARCHAR(100) NOT NULL DEFAULT 'MANUAL',
    source_entity      VARCHAR(100) NOT NULL DEFAULT 'DEFAULT',

    CONSTRAINT chk_dim_customers_scd_is_active
        CHECK (is_active IN ('Y', 'N'))
);

CREATE UNIQUE INDEX IF NOT EXISTS ux_dim_customers_scd_src_period
    ON bl_dm.dim_customers_scd
    (
        COALESCE(customer_src_id, 'n.a.'),
        start_dt,
        COALESCE(source_system, 'n.a.'),
        COALESCE(source_entity, 'n.a.')
    );

/* ============================================================
   9. FCT_CUSTOMER_SATISFACTIONS_DD
   ============================================================ */

CREATE TABLE IF NOT EXISTS bl_dm.fct_customer_satisfactions_dd
(
    event_dt                                      DATE NOT NULL DEFAULT DATE '1900-01-01',
    customer_surr_id                             BIGINT NOT NULL DEFAULT -1,
    travel_surr_id                               BIGINT NOT NULL DEFAULT -1,
    satisfaction_status_surr_id                  BIGINT NOT NULL DEFAULT -1,

    customer_satisfaction_src_id                 VARCHAR(255) NOT NULL DEFAULT 'n.a.',

    fct_inflight_wifi_service                    NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_departure_arrival_time_convenient        NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_ease_of_online_booking                   NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_gate_location                            NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_food_and_drink                           NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_online_boarding                          NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_seat_comfort                             NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_inflight_entertainment                   NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_on_board_service                         NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_leg_room_service                         NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_baggage_handling                         NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_checkin_service                          NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_inflight_service                         NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_cleanliness                              NUMERIC(5,2) NOT NULL DEFAULT 0,
    fct_departure_delay_in_minutes               NUMERIC(10,2) NOT NULL DEFAULT 0,
    fct_arrival_delay_in_minutes                 NUMERIC(10,2) NOT NULL DEFAULT 0,

    ta_insert_dt                                 TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ta_update_dt                                 TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_fct_customer_satisfactions_time_day
        FOREIGN KEY (event_dt)
        REFERENCES bl_dm.dim_time_day (event_dt),

    CONSTRAINT fk_fct_customer_satisfactions_customer
        FOREIGN KEY (customer_surr_id)
        REFERENCES bl_dm.dim_customers_scd (customer_surr_id),

    CONSTRAINT fk_fct_customer_satisfactions_travel
        FOREIGN KEY (travel_surr_id)
        REFERENCES bl_dm.dim_travels (travel_surr_id),

    CONSTRAINT fk_fct_customer_satisfactions_status
        FOREIGN KEY (satisfaction_status_surr_id)
        REFERENCES bl_dm.dim_satisfaction_statuses (satisfaction_status_surr_id)
);

CREATE UNIQUE INDEX IF NOT EXISTS ux_fct_customer_satisfactions_src
    ON bl_dm.fct_customer_satisfactions_dd
    (
        COALESCE(customer_satisfaction_src_id, 'n.a.')
    );

CREATE INDEX IF NOT EXISTS ix_fct_customer_satisfactions_event_dt
    ON bl_dm.fct_customer_satisfactions_dd (event_dt);

CREATE INDEX IF NOT EXISTS ix_fct_customer_satisfactions_customer
    ON bl_dm.fct_customer_satisfactions_dd (customer_surr_id);


/* ============================================================
   10. FCT_BOOKINGS_DD
   ============================================================ */

CREATE TABLE IF NOT EXISTS bl_dm.fct_bookings_dd
(
    event_dt                       DATE NOT NULL DEFAULT DATE '1900-01-01',
    customer_surr_id              BIGINT NOT NULL DEFAULT -1,
    travel_surr_id                BIGINT NOT NULL DEFAULT -1,
    booking_channel_surr_id       BIGINT NOT NULL DEFAULT -1,
    payment_method_surr_id        BIGINT NOT NULL DEFAULT -1,
    loyalty_surr_id               BIGINT NOT NULL DEFAULT -1,
    travel_insurance_surr_id      BIGINT NOT NULL DEFAULT -1,

    booking_src_id                VARCHAR(255) NOT NULL DEFAULT 'n.a.',

    fct_ticket_price              NUMERIC(18,4) NOT NULL DEFAULT 0,
    fct_operating_cost            NUMERIC(18,4) NOT NULL DEFAULT 0,
    fct_airport_fee               NUMERIC(18,4) NOT NULL DEFAULT 0,
    fct_promotion_discount        NUMERIC(18,4) NOT NULL DEFAULT 0,
    fct_profit                    NUMERIC(18,4) NOT NULL DEFAULT 0,

    ta_insert_dt                  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    ta_update_dt                  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_fct_bookings_time_day
        FOREIGN KEY (event_dt)
        REFERENCES bl_dm.dim_time_day (event_dt),

    CONSTRAINT fk_fct_bookings_customer
        FOREIGN KEY (customer_surr_id)
        REFERENCES bl_dm.dim_customers_scd (customer_surr_id),

    CONSTRAINT fk_fct_bookings_travel
        FOREIGN KEY (travel_surr_id)
        REFERENCES bl_dm.dim_travels (travel_surr_id),

    CONSTRAINT fk_fct_bookings_booking_channel
        FOREIGN KEY (booking_channel_surr_id)
        REFERENCES bl_dm.dim_booking_channels (booking_channel_surr_id),

    CONSTRAINT fk_fct_bookings_payment_method
        FOREIGN KEY (payment_method_surr_id)
        REFERENCES bl_dm.dim_payment_methods (payment_method_surr_id),

    CONSTRAINT fk_fct_bookings_loyalty
        FOREIGN KEY (loyalty_surr_id)
        REFERENCES bl_dm.dim_loyalties (loyalty_surr_id),

    CONSTRAINT fk_fct_bookings_travel_insurance
        FOREIGN KEY (travel_insurance_surr_id)
        REFERENCES bl_dm.dim_travel_insurances (travel_insurance_surr_id)
);

CREATE UNIQUE INDEX IF NOT EXISTS ux_fct_bookings_src
    ON bl_dm.fct_bookings_dd
    (
        COALESCE(booking_src_id, 'n.a.')
    );

CREATE INDEX IF NOT EXISTS ix_fct_bookings_event_dt
    ON bl_dm.fct_bookings_dd (event_dt);

CREATE INDEX IF NOT EXISTS ix_fct_bookings_customer
    ON bl_dm.fct_bookings_dd (customer_surr_id);

CREATE INDEX IF NOT EXISTS ix_fct_bookings_travel
    ON bl_dm.fct_bookings_dd (travel_surr_id);
