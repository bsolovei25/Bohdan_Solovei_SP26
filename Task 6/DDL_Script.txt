CREATE SCHEMA IF NOT EXISTS BL_3NF;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_CUSTOMER_ID START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_TRAVEL_ID START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_SATISFACTION_STATUS_ID START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_BOOKING_CHANNEL_ID START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_PAYMENT_METHOD_ID START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_LOYALTY_ID START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_TRAVEL_INSURANCE_ID START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_CUSTOMER_SATISFACTION_ID START WITH 1 INCREMENT BY 1;
CREATE SEQUENCE IF NOT EXISTS BL_3NF.SEQ_BOOKING_ID START WITH 1 INCREMENT BY 1;

-- Table 1: CE_CUSTOMERS_SCD (Slowly Changing Dimension Type 2)
CREATE TABLE IF NOT EXISTS BL_3NF.CE_CUSTOMERS_SCD (
    CUSTOMER_ID         NUMERIC PRIMARY KEY,
    CUSTOMER_SRC_ID     VARCHAR(100) NOT NULL,
    GENDER              VARCHAR(50),
    AGE                 INTEGER,
    CUSTOMER_TYPE       VARCHAR(100),
    START_DT            DATE NOT NULL,
    END_DT              DATE NOT NULL,
    IS_ACTIVE           VARCHAR(1) NOT NULL,
    INSERT_DT           TIMESTAMP NOT NULL,
    SOURCE_SYSTEM       VARCHAR(100) NOT NULL,
    SOURCE_ENTITY       VARCHAR(100) NOT NULL
);

-- Table 2: CE_TRAVELS
CREATE TABLE IF NOT EXISTS BL_3NF.CE_TRAVELS (
    TRAVEL_ID           NUMERIC PRIMARY KEY,
    TRAVEL_SRC_ID       VARCHAR(100) NOT NULL,
    TRAVEL_TYPE         VARCHAR(100),
    TRAVEL_CLASS        VARCHAR(100),
    FLIGHT_DISTANCE     NUMERIC,
    INSERT_DT           TIMESTAMP NOT NULL,
    UPDATE_DT           TIMESTAMP NOT NULL,
    SOURCE_SYSTEM       VARCHAR(100) NOT NULL,
    SOURCE_ENTITY       VARCHAR(100) NOT NULL
);

-- Table 3: CE_SATISFACTION_STATUSES
CREATE TABLE IF NOT EXISTS BL_3NF.CE_SATISFACTION_STATUSES (
    SATISFACTION_STATUS_ID   NUMERIC PRIMARY KEY,
    SATISFACTION_STATUS_SRC_ID VARCHAR(100) NOT NULL,
    SATISFACTION_STATUS      VARCHAR(100),
    INSERT_DT                TIMESTAMP NOT NULL,
    UPDATE_DT                TIMESTAMP NOT NULL,
    SOURCE_SYSTEM            VARCHAR(100) NOT NULL,
    SOURCE_ENTITY            VARCHAR(100) NOT NULL
);

-- Table 4: CE_BOOKING_CHANNELS
CREATE TABLE IF NOT EXISTS BL_3NF.CE_BOOKING_CHANNELS (
    BOOKING_CHANNEL_ID       NUMERIC PRIMARY KEY,
    BOOKING_CHANNEL_SRC_ID   VARCHAR(100) NOT NULL,
    BOOKING_CHANNEL          VARCHAR(100),
    INSERT_DT                TIMESTAMP NOT NULL,
    UPDATE_DT                TIMESTAMP NOT NULL,
    SOURCE_SYSTEM            VARCHAR(100) NOT NULL,
    SOURCE_ENTITY            VARCHAR(100) NOT NULL
);

-- Table 5: CE_PAYMENT_METHODS
CREATE TABLE IF NOT EXISTS BL_3NF.CE_PAYMENT_METHODS (
    PAYMENT_METHOD_ID       NUMERIC PRIMARY KEY,
    PAYMENT_METHOD_SRC_ID   VARCHAR(100) NOT NULL,
    PAYMENT_METHOD          VARCHAR(100),
    INSERT_DT               TIMESTAMP NOT NULL,
    UPDATE_DT               TIMESTAMP NOT NULL,
    SOURCE_SYSTEM           VARCHAR(100) NOT NULL,
    SOURCE_ENTITY           VARCHAR(100) NOT NULL
);

-- Table 6: CE_LOYALTIES
CREATE TABLE IF NOT EXISTS BL_3NF.CE_LOYALTIES (
    LOYALTY_ID              NUMERIC PRIMARY KEY,
    LOYALTY_SRC_ID          VARCHAR(100) NOT NULL,
    LOYALTY_LEVEL           VARCHAR(100),
    INSERT_DT               TIMESTAMP NOT NULL,
    UPDATE_DT               TIMESTAMP NOT NULL,
    SOURCE_SYSTEM           VARCHAR(100) NOT NULL,
    SOURCE_ENTITY           VARCHAR(100) NOT NULL
);

-- Table 7: CE_TRAVEL_INSURANCES
CREATE TABLE IF NOT EXISTS BL_3NF.CE_TRAVEL_INSURANCES (
    TRAVEL_INSURANCE_ID     NUMERIC PRIMARY KEY,
    TRAVEL_INSURANCE_SRC_ID VARCHAR(100) NOT NULL,
    TRAVEL_INSURANCE        VARCHAR(100),
    INSERT_DT               TIMESTAMP NOT NULL,
    UPDATE_DT               TIMESTAMP NOT NULL,
    SOURCE_SYSTEM           VARCHAR(100) NOT NULL,
    SOURCE_ENTITY           VARCHAR(100) NOT NULL
);


-- Table 8: CE_CUSTOMER_SATISFACTIONS
CREATE TABLE IF NOT EXISTS BL_3NF.CE_CUSTOMER_SATISFACTIONS (
    CUSTOMER_SATISFACTION_ID          NUMERIC PRIMARY KEY,
    CUSTOMER_ID                       NUMERIC NOT NULL,
    TRAVEL_ID                         NUMERIC NOT NULL,
    SATISFACTION_STATUS_ID            NUMERIC NOT NULL,
    CUSTOMER_SATISFACTION_SRC_ID      VARCHAR(100) NOT NULL,
    SURVEY_DT                         DATE,
    INFLIGHT_WIFI_SERVICE             NUMERIC,
    DEPARTURE_ARRIVAL_TIME_CONVENIENT NUMERIC,
    EASE_OF_ONLINE_BOOKING            NUMERIC,
    GATE_LOCATION                     NUMERIC,
    FOOD_AND_DRINK                    NUMERIC,
    ONLINE_BOARDING                   NUMERIC,
    SEAT_COMFORT                      NUMERIC,
    INFLIGHT_ENTERTAINMENT            NUMERIC,
    ON_BOARD_SERVICE                  NUMERIC,
    LEG_ROOM_SERVICE                  NUMERIC,
    BAGGAGE_HANDLING                  NUMERIC,
    CHECKIN_SERVICE                   NUMERIC,
    INFLIGHT_SERVICE                  NUMERIC,
    CLEANLINESS                       NUMERIC,
    DEPARTURE_DELAY_IN_MINUTES        NUMERIC,
    ARRIVAL_DELAY_IN_MINUTES          NUMERIC,
    INSERT_DT                         TIMESTAMP NOT NULL,
    UPDATE_DT                         TIMESTAMP NOT NULL,
    SOURCE_SYSTEM                     VARCHAR(100) NOT NULL,
    SOURCE_ENTITY                     VARCHAR(100) NOT NULL,

    CONSTRAINT FK_CS_CUSTOMER FOREIGN KEY (CUSTOMER_ID)
        REFERENCES BL_3NF.CE_CUSTOMERS_SCD(CUSTOMER_ID),
    CONSTRAINT FK_CS_TRAVEL FOREIGN KEY (TRAVEL_ID)
        REFERENCES BL_3NF.CE_TRAVELS(TRAVEL_ID),
    CONSTRAINT FK_CS_STATUS FOREIGN KEY (SATISFACTION_STATUS_ID)
        REFERENCES BL_3NF.CE_SATISFACTION_STATUSES(SATISFACTION_STATUS_ID)
);

-- Table 9: CE_BOOKINGS
CREATE TABLE IF NOT EXISTS BL_3NF.CE_BOOKINGS (
    BOOKING_ID              NUMERIC PRIMARY KEY,
    CUSTOMER_ID             NUMERIC NOT NULL,
    TRAVEL_ID               NUMERIC NOT NULL,
    BOOKING_CHANNEL_ID      NUMERIC NOT NULL,
    PAYMENT_METHOD_ID       NUMERIC NOT NULL,
    LOYALTY_ID              NUMERIC NOT NULL,
    TRAVEL_INSURANCE_ID     NUMERIC NOT NULL,
    BOOKING_DT              DATE,
    TICKET_PRICE            NUMERIC(10,2),
    OPERATING_COST          NUMERIC(10,2),
    AIRPORT_FEE             NUMERIC(10,2),
    PROMOTION_DISCOUNT      NUMERIC(10,2),
    PROFIT                  NUMERIC(10,2),
    INSERT_DT               TIMESTAMP NOT NULL,
    UPDATE_DT               TIMESTAMP NOT NULL,
    SOURCE_SYSTEM           VARCHAR(100) NOT NULL,
    SOURCE_ENTITY           VARCHAR(100) NOT NULL,

    CONSTRAINT FK_B_CUSTOMER FOREIGN KEY (CUSTOMER_ID)
        REFERENCES BL_3NF.CE_CUSTOMERS_SCD(CUSTOMER_ID),
    CONSTRAINT FK_B_TRAVEL FOREIGN KEY (TRAVEL_ID)
        REFERENCES BL_3NF.CE_TRAVELS(TRAVEL_ID),
    CONSTRAINT FK_B_CHANNEL FOREIGN KEY (BOOKING_CHANNEL_ID)
        REFERENCES BL_3NF.CE_BOOKING_CHANNELS(BOOKING_CHANNEL_ID),
    CONSTRAINT FK_B_PAYMENT FOREIGN KEY (PAYMENT_METHOD_ID)
        REFERENCES BL_3NF.CE_PAYMENT_METHODS(PAYMENT_METHOD_ID),
    CONSTRAINT FK_B_LOYALTY FOREIGN KEY (LOYALTY_ID)
        REFERENCES BL_3NF.CE_LOYALTIES(LOYALTY_ID),
    CONSTRAINT FK_B_INSURANCE FOREIGN KEY (TRAVEL_INSURANCE_ID)
        REFERENCES BL_3NF.CE_TRAVEL_INSURANCES(TRAVEL_INSURANCE_ID)
);
